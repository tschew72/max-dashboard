## 1. Duration Timer Component

### File: `components/flow/DurationTimer.tsx`

A lightweight client component that displays elapsed time from a given start timestamp, updating every second.

```typescript
interface DurationTimerProps {
  startedAt: string  // ISO timestamp
  running: boolean   // if false, show static duration (don't tick)
}
```

#### Implementation

```typescript
'use client'
import { useState, useEffect } from 'react'

function formatDuration(ms: number): string {
  const seconds = Math.floor(ms / 1000)
  if (seconds < 60) return `${seconds}s`
  const minutes = Math.floor(seconds / 60)
  const remainingSeconds = seconds % 60
  if (minutes < 60) return `${minutes}m ${remainingSeconds}s`
  const hours = Math.floor(minutes / 60)
  const remainingMinutes = minutes % 60
  return `${hours}h ${remainingMinutes}m`
}

export default function DurationTimer({ startedAt, running }: DurationTimerProps) {
  const [elapsed, setElapsed] = useState(() => Date.now() - new Date(startedAt).getTime())

  useEffect(() => {
    if (!running) return
    const interval = setInterval(() => {
      setElapsed(Date.now() - new Date(startedAt).getTime())
    }, 1000)
    return () => clearInterval(interval)
  }, [startedAt, running])

  return <span className="duration-timer">{formatDuration(elapsed)}</span>
}
```

#### Styling
- Font: tabular-nums (monospace numbers so width doesn't jump)
- Color: `#94a3b8` (muted)
- Size: 12px

## 2. Time Ago Utility

### File: Add to `lib/flow/parse-sessions.ts` (or `lib/flow/utils.ts`)

```typescript
export function formatTimeAgo(isoDate: string | null): string {
  if (!isoDate) return 'never'
  const diff = Date.now() - new Date(isoDate).getTime()
  const minutes = Math.floor(diff / 60000)
  if (minutes < 1) return 'just now'
  if (minutes < 60) return `${minutes}m ago`
  const hours = Math.floor(minutes / 60)
  if (hours < 24) return `${hours}h ago`
  const days = Math.floor(hours / 24)
  if (days < 7) return `${days}d ago`
  return `${Math.floor(days / 7)}w ago`
}
```

## 3. Mode Transitions

### Active → Overview Transition

When the last active chain completes:
1. Chain steps transition to "DONE" state (green)
2. After 5 seconds of no active chains, fade out ActiveChainView (300ms opacity transition)
3. Fade in SystemOverview (300ms opacity transition)
4. Total transition: feels smooth, not jarring

### Overview → Active Transition

When a new chain appears:
1. Immediately switch to ActiveChainView (no delay — urgency)
2. New chain card fades in (200ms)
3. Overview disappears instantly (users want to see what started)

### Implementation

In `FlowView` (page component):
```typescript
const [fadeState, setFadeState] = useState<'in' | 'out' | 'none'>('none')

// When mode changes:
useEffect(() => {
  if (mode changed to 'overview') {
    setFadeState('out')  // fade out active view
    setTimeout(() => {
      setActualMode('overview')
      setFadeState('in')  // fade in overview
      setTimeout(() => setFadeState('none'), 300)
    }, 300)
  } else if (mode changed to 'active') {
    setActualMode('active')  // instant switch
    setFadeState('in')
    setTimeout(() => setFadeState('none'), 200)
  }
}, [mode])
```
