## Overview

Complete rewrite of `app/(dashboard)/flow/page.tsx`. Removes ReactFlow entirely. Implements a two-mode layout that automatically switches based on whether active spawn chains exist. SSE drives all real-time updates.

## File: `app/(dashboard)/flow/page.tsx`

### Architecture

```
FlowPage (server component wrapper)
└── FlowView (client component)
    ├── ActiveChainView (when activeChains.length > 0)
    └── SystemOverview (when activeChains.length === 0)
```

**SSE subscription:**
```typescript
useEffect(() => {
  let es: EventSource | null = null
  let reconnectTimer: ReturnType<typeof setTimeout>

  function connect() {
    es = new EventSource('/api/flow/stream')
    
    es.onmessage = (event) => {
      try {
        const msg = JSON.parse(event.data)
        if (msg.type === 'state') {
          const newState: FlowState = msg.data
          setState(newState)
          
          // Auto-switch modes
          const shouldBeActive = newState.activeChains.length > 0
          setMode(prev => {
            if (prev === 'active' && !shouldBeActive) {
              // Transition out: brief delay for fade
              setTransitioning(true)
              setTimeout(() => setTransitioning(false), 300)
              return 'overview'
            }
            if (prev === 'overview' && shouldBeActive) {
              setTransitioning(true)
              setTimeout(() => setTransitioning(false), 300)
              return 'active'
            }
            return prev
          })
        }
      } catch { /* ignore parse errors */ }
    }

    es.onerror = () => {
      es?.close()
      reconnectTimer = setTimeout(connect, 5000)
    }
  }

  connect()
  return () => {
    es?.close()
    clearTimeout(reconnectTimer)
  }
}, [])
```

**Fallback poll (belt-and-suspenders):**
```typescript
useEffect(() => {
  const interval = setInterval(async () => {
    try {
      const res = await fetch('/api/flow/agents')
      if (res.ok) {
        const data: FlowState = await res.json()
        setState(data)
        setMode(data.activeChains.length > 0 ? 'active' : 'overview')
      }
    } catch { /* retry next interval */ }
  }, 30_000)  // 30s fallback
  return () => clearInterval(interval)
}, [])
```
